% ===== Document Preamble =====
\documentclass[11pt]{article}
\usepackage[margin=0.5in]{geometry}
\usepackage{amsmath,amssymb}
\usepackage{xcolor}
\usepackage{enumitem}
\usepackage{needspace}
\usepackage{comment}

% ===== Colors =====
\definecolor{AnsRed}{RGB}{170,0,0}
\definecolor{WrongBlue}{RGB}{0,0,160}
\definecolor{SkipGray}{RGB}{120,120,120}

% ===== Choice list styling (tight, consistent) =====
\setlist[enumerate]{%
  leftmargin=1.5em,
  itemsep=0.25em,
  topsep=0.25em,
  label=\textbf{\Alph*.}
}

% ===== Style Wrappers =====
\newcommand{\QuestionStyle}[1]{{\large\bfseries\boldmath #1}}
\newcommand{\AnswerStyle}[1]{{\large\bfseries\boldmath #1}}
\newcommand{\ReasonStyle}[1]{{\normalsize #1}}

% ===== Show / Hide Reasoning Toggle =====
\newif\ifshowreason
\showreasontrue      % Instructor version
% \showreasonfalse   % Student version

% ===== Choice Item Macros (Answer + Reasoning) =====
% Usage:
%   \Correct{Answer}{Reasoning}
%   \Wrong{Answer}{Reasoning}

\newcommand{\Correct}[2]{%
  \item \AnswerStyle{\textcolor{AnsRed}{#1}}%
  \ifshowreason\\[-0.2em]
    \ReasonStyle{\textcolor{AnsRed}{#2}}%
  \fi
}

\newcommand{\Wrong}[2]{%
  \item \AnswerStyle{\textcolor{WrongBlue}{#1}}%
  \ifshowreason\\[-0.2em]
    \ReasonStyle{\textcolor{WrongBlue}{#2}}%
  \fi
}

% ===== Shared-Context Group Preamble Macro =====
% Usage: \GroupPreamble{shared context text}
% Rendered ONCE before the first question of a shared-context group (e.g.
% "For questions 3-7, consider ..."). Theme-neutral; reusable for any shared
% stimulus, passage, table reference, or section intro.

\newcommand{\GroupPreamble}[1]{%
  \Needspace{4\baselineskip}%
  \noindent #1\par
  \vspace{0.5em}
}

% ===== Multiple-Choice Question Block Macro =====
% Usage: \QBlock{Q1}{stem...}{<items>}{answer key}

\newcommand{\QBlock}[4]{%
  \Needspace{8\baselineskip}%
  \noindent
  \begin{minipage}{\linewidth}
    \QuestionStyle{#1.~#2}

    \vspace{0.25em}
    \begin{enumerate}
      #3
    \end{enumerate}

    \vspace{0.1em}
    \noindent\textcolor{AnsRed}{\textit{\textbf{Answer:}~#4}}

    \vspace{0.25em}
    \hrule height 0.4pt
  \end{minipage}
  \vspace{0.75em}
}

% ===== Open-Ended Question Block Macro =====
% Usage: \OEBlock{Q1}{stem...}{answer text}{explanation text}
% Answer renders in red; the supporting explanation renders in blue and is
% hidden when \showreasonfalse is set.

\newcommand{\OEBlock}[4]{%
  \Needspace{5\baselineskip}%
  \noindent
  \begin{minipage}{\linewidth}
    \QuestionStyle{#1.~#2}

    \vspace{0.25em}
    \noindent\textcolor{AnsRed}{\textit{\textbf{Answer:}~#3}}%
    \ifshowreason
      \par\vspace{0.15em}
      \noindent\ReasonStyle{\textcolor{WrongBlue}{\textit{\textbf{Explanation:}}~#4}}%
    \fi

    \vspace{0.25em}
    \hrule height 0.4pt
  \end{minipage}
  \vspace{0.75em}
}

% ===== Multi-Part Question Block Macro =====
% Usage: \MPBlock{Q1}{stem...}{<parts>}

\newcommand{\MPBlock}[3]{%
  \Needspace{8\baselineskip}%
  \noindent
  \begin{minipage}{\linewidth}
    \QuestionStyle{#1.~#2}

    \vspace{0.25em}
    #3

    \vspace{0.25em}
    \hrule height 0.4pt
  \end{minipage}
  \vspace{0.75em}
}

% ===== MC Sub-Part Macro =====
% Usage: \MCPart{a}{stem...}{<items>}{answer key}

\newcommand{\MCPart}[4]{%
  \par\vspace{0.35em}
  \noindent\textbf{(#1)}~#2
  \begin{enumerate}
    #3
  \end{enumerate}
  \vspace{0.1em}
  \noindent\textit{\textbf{Answer:}~#4}
  \par
}

% ===== Open-Ended Sub-Part Macro =====
% Usage: \OEPart{a}{stem...}{answer text}{explanation text}
% Same coloring as \OEBlock: red answer, blue explanation gated by \ifshowreason.

\newcommand{\OEPart}[4]{%
  \par\vspace{0.35em}
  \noindent\textbf{(#1)}~#2
  \par\vspace{0.15em}
  \noindent\textcolor{AnsRed}{\textit{\textbf{Answer:}~#3}}%
  \ifshowreason
    \par\vspace{0.15em}
    \noindent\ReasonStyle{\textcolor{WrongBlue}{\textit{\textbf{Explanation:}}~#4}}%
  \fi
  \par
}

% ===== Skipped Question Note Macro =====
% Usage: \SkipBlock{Q1}{reason}

\newcommand{\SkipBlock}[2]{%
  \noindent\textcolor{SkipGray}{\textit{[#1 omitted: #2]}}%
  \vspace{0.5em}\par
}
